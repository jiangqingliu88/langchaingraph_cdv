# src/core/coordination.py

class CoordinationAgent:
    """负责PPA变更的跨团队协同通知"""
    
    def notify_ppa_change(self, ppa_delta: Dict):
        """
        PPA更新时，差异化通知两个团队
        """
        # 1. 通知验证团队：重新跑回归
        self._notify_verification_team({
            "reason": "PPA变更导致RTL更新",
            "affected_modules": ppa_delta["affected_rtl"],
            "action": "重新跑覆盖率回归",
            "priority": "HIGH"
        })
        
        # 2. 通知设计团队：调整RTL
        self._notify_design_team({
            "reason": "PPA指标调整",
            "new_constraints": ppa_delta["constraints"],
            "action": "优化RTL以满足新PPA指标",
            "deadline": ppa_delta["deadline"]
        })
        
        # 3. 证据链记录
        self._record_evidence(ppa_delta)
```

### 通知方式（按场景区分）

| 通知方式 | 适用场景 | 实现 |
| :--- | :--- | :--- |
| **自动触发 Agent** | PPA变化量小，自动修复 | `design` Agent 自动改RTL，`verify` Agent 自动补测 |
| **审批门控** | PPA变化量大，需人工决策 | 生成 `approval_request.json`，工程师审批后执行 |
| **邮件/IM通知** | 仅告知，无需即时动作 | 通过企业微信/Slack Webhook 发送摘要 |
| **证据链写入** | 所有场景 | 无论哪种通知，都写入 `evidence_chain.json` |

> **第二个闭环是 PPA 闭环**：由 SoC 集成团队驱动，目标是设计竞争力（性能/功耗/面积），迭代频率低（小时/天级）。
> 
> 两个闭环通过 **CoordinationAgent（协同层）** 连接——当 SoC 级 PPA 指标更新时，协同层会**差异化通知两个团队**：
> - 通知验证团队：‘PPA 变了，RTL 有更新，请重新跑覆盖率回归’
> - 通知设计团队：‘PPA 指标调整了，请优化 RTL 以满足新约束’
> 
> 同时，所有通知、决策、修改动作都写入统一的证据链，确保最终签核时‘功能正确’和‘设计竞争力’落在同一份审计报告里。这解决了传统流程中‘验证做完才发现PPA不达标’的痛点——两个闭环不是串行的，而是异步协同的。”

---

## 💎 一句话总结

> **“覆盖率闭环跑‘功能对不对’，PPA闭环跑‘设计好不好’，两个闭环通过协同层异步通信，差异化通知验证和设计两个团队——各自闭环，统一证据。”**
